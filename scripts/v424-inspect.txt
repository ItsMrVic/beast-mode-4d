function deduplicateAll(){
  let removed=0;
  operators.forEach(op=>{
    const seen=new Set();
    const before=op.draws.length;
    op.draws=op.draws.filter(d=>{
      const key=d.date+d.top3.join('');
      if(seen.has(key)){removed++;return false;}

    });
  });
  if(removed>0){saveAll();toast(`🧹 Removed ${removed} duplicate(s)`);render();}
  return removed;
}

function mulberry32(seed){
  return function(){
    let t=(seed+=0x6D2B79F5)|0;
    t=Math.imul(t^t>>>15,t|1);
    t^=t+Math.imul(t^t>>>7,t|61);
    return ((t^t>>>14)>>>0)/4294967296;
  };
}
